"""페이지 공통 도우미: 차트·표 표시, 결과 CSV 읽기"""

import inspect
from pathlib import Path

import pandas as pd
import streamlit as st

BASE_DIR = Path(__file__).resolve().parent
RESULT_DIR = BASE_DIR / "output"


def _stretch_default(fn) -> bool:
    """이 Streamlit 버전에서 width 기본값이 "stretch"인지 확인한다."""
    p = inspect.signature(fn).parameters.get("width")
    return p is not None and p.default == "stretch"


_CHART_NEW = _stretch_default(st.plotly_chart)
_TABLE_NEW = _stretch_default(st.dataframe)


def show_chart(fig) -> None:
    """버전과 무관하게 컨테이너 폭에 맞춰 차트를 그린다."""
    if _CHART_NEW:
        st.plotly_chart(fig)
    else:
        st.plotly_chart(fig, use_container_width=True)


def show_table(df: pd.DataFrame, **kwargs) -> None:
    if _TABLE_NEW:
        st.dataframe(df, hide_index=True, **kwargs)
    else:
        st.dataframe(df, hide_index=True, use_container_width=True, **kwargs)


@st.cache_data
def load_result(name: str) -> pd.DataFrame | None:
    """output/ 폴더의 결과 CSV를 읽는다. 없으면 None."""
    path = RESULT_DIR / name
    if not path.exists():
        return None
    return pd.read_csv(path, encoding="utf-8-sig")


def missing_notice(script: str) -> None:
    st.info(f"결과 파일이 없습니다. `python {script}`를 실행해 만들어진 `output/` 폴더를 함께 올려 주세요.")
